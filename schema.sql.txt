CREATE DATABASE IF NOT EXISTS oms_db;
USE oms_db;

CREATE TABLE IF NOT EXISTS admin (
    admin_id INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(120) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'Staff'
);

CREATE TABLE IF NOT EXISTS children (
    child_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    date_of_birth DATE NOT NULL,
    admission_date DATE NOT NULL,
    residential_status VARCHAR(50) NOT NULL,
    blood_group VARCHAR(5) NOT NULL,
    is_active BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE IF NOT EXISTS staff (
    staff_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    role VARCHAR(50) NOT NULL,
    contact_number VARCHAR(20) NOT NULL,
    schedule VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS donors (
    donor_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    donor_type VARCHAR(50) NOT NULL
);

CREATE TABLE IF NOT EXISTS donations (
    donation_id INT AUTO_INCREMENT PRIMARY KEY,
    donor_id INT NOT NULL,
    amount_or_item VARCHAR(100) NOT NULL,
    donation_date DATE NOT NULL,
    FOREIGN KEY (donor_id) REFERENCES donors(donor_id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS volunteers (
    volunteer_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    assigned_duty VARCHAR(100) NOT NULL
);

CREATE TABLE IF NOT EXISTS inventory (
    item_id INT AUTO_INCREMENT PRIMARY KEY,
    category VARCHAR(50) NOT NULL,
    stock_quantity INT NOT NULL DEFAULT 0,
    reorder_threshold INT NOT NULL DEFAULT 10
);

CREATE TABLE IF NOT EXISTS expenses (
    expense_id INT AUTO_INCREMENT PRIMARY KEY,
    item_id INT NULL,
    staff_id INT NULL,
    amount DECIMAL(10,2) NOT NULL,
    expense_date DATE NOT NULL,
    FOREIGN KEY (item_id) REFERENCES inventory(item_id) ON DELETE SET NULL,
    FOREIGN KEY (staff_id) REFERENCES staff(staff_id) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS medical_records (
    record_id INT AUTO_INCREMENT PRIMARY KEY,
    child_id INT NOT NULL,
    diagnosis TEXT NOT NULL,
    treatment TEXT NOT NULL,
    visit_date DATE NOT NULL,
    FOREIGN KEY (child_id) REFERENCES children(child_id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS education_records (
    record_id INT AUTO_INCREMENT PRIMARY KEY,
    child_id INT NOT NULL,
    subject VARCHAR(50) NOT NULL,
    grade VARCHAR(10) NOT NULL,
    term VARCHAR(20) NOT NULL,
    FOREIGN KEY (child_id) REFERENCES children(child_id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS visitors (
    visitor_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    purpose VARCHAR(255) NOT NULL,
    check_in_time DATETIME NOT NULL
);

CREATE TABLE IF NOT EXISTS adoptions (
    adoption_id INT AUTO_INCREMENT PRIMARY KEY,
    child_id INT NOT NULL,
    verification_status VARCHAR(50) NOT NULL DEFAULT 'Pending',
    finalization_date DATE NULL,
    FOREIGN KEY (child_id) REFERENCES children(child_id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS events (
    event_id INT AUTO_INCREMENT PRIMARY KEY,
    event_name VARCHAR(100) NOT NULL,
    event_date DATE NOT NULL
);

CREATE TABLE IF NOT EXISTS child_event_participation (
    child_id INT NOT NULL,
    event_id INT NOT NULL,
    participation_notes TEXT NULL,
    PRIMARY KEY (child_id, event_id),
    FOREIGN KEY (child_id) REFERENCES children(child_id) ON DELETE CASCADE,
    FOREIGN KEY (event_id) REFERENCES events(event_id) ON DELETE CASCADE
);

-- DATABASE TRIGGER FOR AUTO-LOGGING INVENTORY DEDUCTION
DELIMITER //
CREATE TRIGGER trg_after_expense_inventory_update
AFTER INSERT ON expenses
FOR EACH ROW
BEGIN
    IF NEW.item_id IS NOT NULL THEN
        UPDATE inventory 
        SET stock_quantity = stock_quantity - 1 
        WHERE item_id = NEW.item_id;
    END IF;
END;
//
DELIMITER ;
